  message(FATAL_ERROR "VSim currently supports Lua 5.4 only")

add_library(vsim_options INTERFACE)
target_compile_features(vsim_options INTERFACE cxx_std_20)
if(MSVC)
  if(VSIM_SANITIZERS)
    message(FATAL_ERROR "The ASan/UBSan preset currently requires GCC or Clang")
  endif()
  target_compile_options(vsim_options INTERFACE /W4 /permissive-)
  if(VSIM_WARNINGS_AS_ERRORS)
    target_compile_options(vsim_options INTERFACE /WX)
  endif()
else()
  target_compile_options(vsim_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(VSIM_WARNINGS_AS_ERRORS)
    target_compile_options(vsim_options INTERFACE -Werror)
  endif()
  if(VSIM_SANITIZERS)
    target_compile_options(vsim_options INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    target_link_options(vsim_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()

add_library(vsim_core STATIC src/scene.cpp src/script.cpp src/clock.cpp src/io.cpp src/physics.cpp src/simulation.cpp src/mesh.cpp)
add_library(vsim::core ALIAS vsim_core)
set_target_properties(vsim_core PROPERTIES CXX_EXTENSIONS OFF EXPORT_NAME core) # Installed as vsim::core, like the alias.
target_include_directories(vsim_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(vsim_core SYSTEM PRIVATE ${LUA_INCLUDE_DIR})
# Warning flags stay in this tree; SDK consumers only inherit the link dependencies.
target_link_libraries(vsim_core PUBLIC glm::glm nlohmann_json::nlohmann_json PRIVATE ${LUA_LIBRARIES} $<BUILD_INTERFACE:vsim_options> Jolt)
target_compile_features(vsim_core PUBLIC cxx_std_20)

if(VSIM_INSTALL_SDK)
  # Headless SDK: `find_package(vsim CONFIG)` gives vsim::core (with vsim::Jolt, glm,
  # nlohmann_json, and Lua as link dependencies); `pkg-config vsim` gives the same flags.
  include(CMakePackageConfigHelpers)
  install(TARGETS vsim_core Jolt EXPORT vsimTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(DIRECTORY include/vsim DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(EXPORT vsimTargets NAMESPACE vsim:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vsim)
  # glm keeps its include directory on glm::glm-header-only and links a small static library.
  set(VSIM_GLM_INCLUDE_DIRS "")
  foreach(candidate IN ITEMS glm::glm-header-only glm::glm)
    if(TARGET ${candidate})
      get_target_property(candidate_dirs ${candidate} INTERFACE_INCLUDE_DIRECTORIES)
      if(candidate_dirs)
        list(APPEND VSIM_GLM_INCLUDE_DIRS ${candidate_dirs})
      endif()
    endif()
  endforeach()
  set(VSIM_GLM_LIBRARY "")
  get_target_property(glm_type glm::glm TYPE)
  if(NOT glm_type STREQUAL "INTERFACE_LIBRARY")
    foreach(property IN ITEMS IMPORTED_LOCATION_RELEASE IMPORTED_LOCATION_NOCONFIG IMPORTED_LOCATION)
      get_target_property(glm_location glm::glm ${property})
      if(glm_location AND NOT VSIM_GLM_LIBRARY)
        set(VSIM_GLM_LIBRARY "${glm_location}")
      endif()
    endforeach()
  endif()
  get_target_property(VSIM_JSON_INCLUDE_DIRS nlohmann_json::nlohmann_json INTERFACE_INCLUDE_DIRECTORIES)
  list(REMOVE_DUPLICATES VSIM_GLM_INCLUDE_DIRS)
  list(REMOVE_DUPLICATES VSIM_JSON_INCLUDE_DIRS)
  set(VSIM_LUA_LIBRARIES "${LUA_LIBRARIES}")
  configure_package_config_file(cmake/vsimConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/vsimConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vsim
    PATH_VARS CMAKE_INSTALL_INCLUDEDIR CMAKE_INSTALL_LIBDIR)
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/vsimConfigVersion.cmake" COMPATIBILITY SameMajorVersion)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vsimConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/vsimConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vsim)
  set(VSIM_PC_INCLUDES "")
  foreach(dir IN LISTS VSIM_GLM_INCLUDE_DIRS VSIM_JSON_INCLUDE_DIRS)
    string(APPEND VSIM_PC_INCLUDES " -I${dir}")
  endforeach()
  set(VSIM_PC_LUA "")
  if(VSIM_GLM_LIBRARY)
    string(APPEND VSIM_PC_LUA " ${VSIM_GLM_LIBRARY}")
  endif()
  foreach(library IN LISTS VSIM_LUA_LIBRARIES)
    if(IS_ABSOLUTE "${library}")
      get_filename_component(library_dir "${library}" DIRECTORY)
      string(APPEND VSIM_PC_LUA " ${library} -Wl,-rpath,${library_dir}")
    else()
      string(APPEND VSIM_PC_LUA " -l${library}")
    endif()
  endforeach()
  configure_file(cmake/vsim.pc.in "${CMAKE_CURRENT_BINARY_DIR}/vsim.pc" @ONLY)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vsim.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
endif()

if(VSIM_BUILD_3D)
  find_package(glfw3 3.4 CONFIG REQUIRED)
  find_package(Vulkan 1.3 REQUIRED)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  find_program(VSIM_GLSLC glslc REQUIRED)
  set(shader_binaries "")
  foreach(shader IN ITEMS scene.vert scene.frag shadow.vert fullscreen.vert tonemap.frag ssao.frag ssao_blur.frag overlay.vert overlay.frag)
    set(binary "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.spv")
    add_custom_command(OUTPUT "${binary}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
      COMMAND ${VSIM_GLSLC} --target-env=vulkan1.3 "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" -o "${binary}"
      DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" VERBATIM)
    list(APPEND shader_binaries "${binary}")
  endforeach()
  set(shader_header "${CMAKE_CURRENT_BINARY_DIR}/generated/scene_shaders.hpp")
  add_custom_command(OUTPUT "${shader_header}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_shaders.py" "${shader_header}" ${shader_binaries}
    DEPENDS ${shader_binaries} scripts/embed_shaders.py VERBATIM)
  add_library(vsim_renderer STATIC src/renderer.cpp src/overlay.cpp "${shader_header}")
  target_include_directories(vsim_renderer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(vsim_renderer PUBLIC vsim_core PRIVATE nvrhi_vk glfw Vulkan::Headers Vulkan::Vulkan vsim_options)
  add_executable(vsim_game apps/game.cpp)
  target_link_libraries(vsim_game PRIVATE vsim_renderer vsim_options)
  set_target_properties(vsim_game PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
  install(TARGETS vsim_game RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES examples/crystal_courier_3d/scene.json DESTINATION ${CMAKE_INSTALL_DATADIR}/vsim/examples/crystal_courier_3d)
  install(FILES "${VSIM_NVRHI_SOURCE}/LICENSE.txt" "${VSIM_NVRHI_SOURCE}/ThirdPartyLicenses.txt" DESTINATION ${CMAKE_INSTALL_DATADIR}/vsim/licenses/NVRHI)
  install(FILES "${VSIM_VULKAN_HEADERS_SOURCE}/LICENSE.md" DESTINATION ${CMAKE_INSTALL_DATADIR}/vsim/licenses/Vulkan-Headers)
  install(DIRECTORY "${VSIM_VULKAN_HEADERS_SOURCE}/LICENSES" DESTINATION ${CMAKE_INSTALL_DATADIR}/vsim/licenses/Vulkan-Headers)
endif()

    set(binary "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.spv")
    add_custom_command(OUTPUT "${binary}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"

  endforeach()
  set(gui_header "${CMAKE_CURRENT_BINARY_DIR}/generated/gui_shaders.hpp")
  add_custom_command(OUTPUT "${gui_header}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_shaders.py" "${gui_header}" ${gui_binaries}
    DEPENDS ${gui_binaries} scripts/embed_shaders.py VERBATIM)
  add_library(vsim_editor_ui STATIC src/editor_ui.cpp src/gui_renderer.cpp "${gui_header}")
  target_include_directories(vsim_editor_ui PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(vsim_editor_ui PRIVATE GLM_ENABLE_EXPERIMENTAL)
  target_link_libraries(vsim_editor_ui PUBLIC vsim_editor_core vsim_renderer PRIVATE vsim_imgui nvrhi glfw vsim_options)
  add_executable(vsim_editor apps/editor_window.cpp)
  target_link_libraries(vsim_editor PRIVATE vsim_editor_ui vsim_options)
